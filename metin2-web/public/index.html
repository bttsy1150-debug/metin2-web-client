Metin2 2D Server [Lv. ${playerStats.level}]
HP: ${playerStats.hp}/${playerStats.maxHp}

WASD: Haritada Yürü
Boşluk (Space): Yakındaki Hayvana Vur!

50 : 45; // Mesafe eşiği // Eğer oyuncu canavara çok yakınsa canavar saldırmaya başlar if (Math.abs(distX) < 150 && Math.abs(distY) < 150) { if (monster.x < playerPos.x) monster.x += 1; else monster.x -= 1; if (monster.y < playerPos.y) monster.y += 1; else monster.y -= 1; // Çok yakındaysa oyuncunun canını azaltır if (Math.abs(distX) < 20 && Math.abs(distY) < 20) { if (Math.random() < 0.02) { // Rastgele vuruş sıklığı playerStats.hp -= 5; if (playerStats.hp <= 0) { playerStats.hp = playerStats.maxHp; // Ölüp dirilme testi playerPos = { x: 220, y: 180 }; } } } } }); updatePlayerPosition(); drawMonsters(); requestAnimationFrame(gameLoop); } // Oyunu Başlat updateUI(); drawMonsters(); gameLoop();
